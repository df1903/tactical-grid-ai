\chapter{Referencia Rápida de Algoritmos}
\label{ap:algoritmos}
\begin{table}[H]
\centering
\caption{Resumen de algoritmos implementados}
\begin{tabular}{lccc}
\toprule
Algoritmo & Tipo & Complejidad & Óptimo \\
\midrule
BFS & No informada & $\mathcal{O}(V)$ & Solo en movimientos \\
DFS & No informada & $\mathcal{O}(V)$ & No \\
UCS & Coste uniforme & $\mathcal{O}(V \log V)$ & Sí (costo) \\
A* (h1/h2) & Informada & $\mathcal{O}(V \log V)$ & Sí (con h admisible) \\
Beam (k) & Memoria limitada & $\mathcal{O}(L \cdot k \cdot b \log(kb))$ & \textbf{No} (incompleto) \\
Minimax & Adversarial & $\mathcal{O}(B^D)$ & Sí (horizonte) \\
Alpha-beta & Adversarial & $\mathcal{O}(B^{D/2})$ mejor & Sí \\
\bottomrule
\end{tabular}
\end{table}
